Claude Code

{% blocktranslate %}Run this once in your terminal, then type /mcp in Claude Code and choose Authenticate to sign in. Claude can then use falcondatabase in every session.{% endblocktranslate %}

claude mcp add --transport http falcondatabase {{ mcp_url }}

{% translate "Or with a token, without signing in:" %}

claude mcp add --transport http falcondatabase {{ mcp_url }} --header "Authorization: Bearer {{ ai_token }}"

Cursor · Windsurf · {% translate "other MCP clients" %}

{% blocktranslate %}Add the server to the client's MCP configuration (in Cursor, .cursor/mcp.json). In VS Code use the key "servers" and add "type": "http".{% endblocktranslate %}

{
  "mcpServers": {
    "falcondatabase": {
      "url": "{{ mcp_url }}",
      "headers": { "Authorization": "Bearer {{ ai_token }}" }
    }
  }
}

Claude Desktop ({% translate "with a token" %})

{% blocktranslate %}In Settings → Developer → Edit config (claude_desktop_config.json). It needs Node.js installed.{% endblocktranslate %}

{
  "mcpServers": {
    "falcondatabase": {
      "command": "npx",
      "args": ["-y", "mcp-remote", "{{ mcp_url }}", "--header", "Authorization:${AUTH_HEADER}"],
      "env": { "AUTH_HEADER": "Bearer {{ ai_token }}" }
    }
  }
}
