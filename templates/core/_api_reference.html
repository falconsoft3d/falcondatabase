{% translate "JavaScript SDK" %}

{% translate "A single file, no dependencies or build step. Import it straight from your server:" %}

import { createClient } from "{{ base_url }}/sdk/falcondb.js"

const db = createClient("{{ base_url }}", "{{ ref }}", "{{ anon_key }}")

// {% translate "Chainable queries" %}
const { data, count, error } = await db.from("products")
  .select("id,name,price")
  .gte("price", 10)
  .ilike("name", "%coffee%")
  .order("price", { desc: true })
  .range(0, 19)

// {% translate "Write" %}
await db.from("products").insert({ name: "Coffee", price: 2.5 })
await db.from("products").update(7, { price: 3 })
await db.from("products").delete(7)

// {% translate "Login (the token is stored in localStorage and sent automatically)" %}
await db.auth.signUp({ email, password })
await db.auth.signIn({ email, password })
const { user } = await db.auth.getUser()
db.auth.signOut()
